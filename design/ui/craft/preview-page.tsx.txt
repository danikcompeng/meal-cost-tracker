function food(id: string, name: string, unit: Food["unit"], kcal: number, protein: number, carbs: number, fat: number, price: number, size: number, water = false): Food {
  return {
    id, name, brand: null, barcode: null, unit, price, package_size: size,
    kcal, protein_g: protein, carbs_g: carbs, fat_g: fat, ...NO_EXTRAS,
    counts_toward_hydration: water,
  };
}

  id, name, servings, ingredients: items.map(([f, amount]) => ({ food: f, amount })),
});

});
const entries: MealEntry[] = [
